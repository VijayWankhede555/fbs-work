Q8)   Create employees Table with Normal Foreign Key 
      Write an SQL query to create a table named employees including columns: 
      employee_id, first_name, last_name, email, phone_number, hire_date, job_id, 
      salary, commission, manager_id and department_id.

answer=      create table employees
           (emp_id int primary key,frist_name varchar(50),last_name varchar(50),email varchar(50)
            ,phone_no varchar(12),
           hire_date date,job_id int,salary decimal(7,2),commision decimal(7,2),
           manager_id int,department_id int  FOREIGN KEY (department_id)
           REFERENCES departments(department_id);
